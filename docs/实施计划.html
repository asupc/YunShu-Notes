2. 数据库完整 DDL 与无限层级树原子移动

CREATE EXTENSION IF NOT EXISTS ltree;
CREATE EXTENSION IF NOT EXISTS pgcrypto;

CREATE TABLE nodes (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL,
  parent_id       UUID REFERENCES nodes(id) ON DELETE CASCADE,
  path            LTREE NOT NULL,            -- 例: root.n_9f1a.n_2b3c
  rank            TEXT NOT NULL,             -- LexoRank 排序键，O(1) 拖拽插入
  title           TEXT NOT NULL,
  node_type       TEXT NOT NULL CHECK (node_type IN ('folder','note','markdown','docx','xlsx','pdf')),
  yjs_state       BYTEA,                     -- note/markdown 节点的 Yjs 二进制合并快照
  current_file_id UUID,                      -- 文件节点指向最新 node_files.id
  current_version INT DEFAULT 1,             -- 文件乐观锁版本号
  plain_text      TEXT DEFAULT '',           -- 抽取的纯文本（含 Office 文档内容）
  is_deleted      BOOLEAN DEFAULT FALSE,
  created_at      TIMESTAMPTZ DEFAULT now(),
  updated_at      TIMESTAMPTZ DEFAULT now()
);
CREATE INDEX idx_nodes_path_gist ON nodes USING GIST (path);
CREATE INDEX idx_nodes_parent_rank ON nodes (user_id, parent_id, rank) WHERE is_deleted = FALSE;

CREATE TABLE node_files (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  node_id         UUID NOT NULL REFERENCES nodes(id) ON DELETE CASCADE,
  version         INT NOT NULL,
  file_name       TEXT NOT NULL,
  mime_type       TEXT NOT NULL,
  file_size       BIGINT NOT NULL,
  sha256          CHAR(64) NOT NULL,
  storage_key     TEXT NOT NULL,
  preview_pdf_key TEXT,
  preview_status  TEXT NOT NULL DEFAULT 'pending', -- pending | ready | failed
  source_device   TEXT,
  created_at      TIMESTAMPTZ DEFAULT now(),
  UNIQUE (node_id, version)
);

CREATE TABLE import_jobs (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL,
  source_type     TEXT NOT NULL CHECK (source_type IN ('graph_api','onepkg_local')),
  target_node_id  UUID REFERENCES nodes(id) ON DELETE SET NULL,
  status          TEXT NOT NULL CHECK (status IN ('queued','running','paused','done','failed')),
  total_pages     INT DEFAULT 0,
  imported_pages  INT DEFAULT 0,
  failed_pages    INT DEFAULT 0,
  cursor_state    JSONB DEFAULT '{}'::jsonb,
  error_log       JSONB DEFAULT '[]'::jsonb,
  updated_at      TIMESTAMPTZ DEFAULT now()
);

原子移动整棵子树 SQL（防循环引用）

-- 1. 前置校验：禁止移动到自身或其后代节点下
SELECT 1 FROM nodes WHERE id = :newParentId AND path <@ (SELECT path FROM nodes WHERE id = :nodeId);

-- 2. 单条事务语句重写整棵子树 ltree 路径
UPDATE nodes
SET path = :newParentPath || subpath(path, nlevel(:oldPath) - 1),
    parent_id = CASE WHEN id = :nodeId THEN :newParentId ELSE parent_id END,
    rank = CASE WHEN id = :nodeId THEN :newRank ELSE rank END,
    updated_at = now()
WHERE user_id = :userId AND path <@ :oldPath;

3. 服务端 REST API 与 WebSocket 协议契约

接口分组 方法与路径 核心参数 / 响应 核心逻辑与异常保护
目录树同步GET /api/v1/nodes/treeQuery: since；返回 NodeMeta[]仅传元数据（不含正文），万级节点压缩后 < 2 MB，支持差量同步。
创建 / 移动POST /api/v1/nodes
PATCH /api/v1/nodes/:id/move		{ parentId, title, nodeType, prevRank, nextRank }服务端计算 ltree 与中间 rank；移动前校验无环。
预签名上传POST /api/v1/files/presign{ nodeId, fileName, fileSize, sha256 }先查 sha256 支持秒传去重，否则签发 10 分钟有效的 S3 PUT 链接。
版本提交POST /api/v1/files/commit{ nodeId, baseVersion, storageKey, sha256, deviceName }校验 current_version === baseVersion；冲突返回 409 并创建冲突副本子节点。
预览与下载GET /api/v1/nodes/:id/file-urls返回 downloadUrl, previewPdfUrl, previewStatus, version签发 15 分钟有效的私有桶签名链接，保障文件不外泄。
全局搜索GET /api/v1/searchQuery: q, types, rootId联合命中标题、笔记正文与 Word/Excel 抽取文本，返回高亮片段与路径。
Hocuspocus 实时协同协议：WebSocket 房间名为 node:<uuid>，连接时在 onAuthenticate 校验 JWT 与节点归属；编辑停止 2 秒防抖后将合并后的 Yjs 二进制快照写回 nodes.yjs_state，并同步更新 nodes.plain_text。

4. PC 端 Rust 本地 Office 唤起与防抖自动回传状态机

  • 独立隔离目录：缓存在 {app_cache_dir}/office_workspace/{node_id}/{file_name}，杜绝同名文件互相覆盖。
  • 必须监听父目录：Word/Excel 保存时采用「写临时文件再原子替换 inode」，直接监听单文件句柄会在首次保存后失效；须监听 {node_id}/ 目录并过滤 ~$* 与 .tmp。
  • 共享读锁探测：1200ms 防抖后若 Windows 报 ERROR_SHARING_VIOLATION，间隔 500ms 最多重试 5 次再算 SHA-256 上传。

5. OneNote 一次性全量迁移深度算法

  • 通道 A（Microsoft Graph API）：按 Notebooks → SectionGroups → Sections → Pages 递归遍历（Graph OneNote API）。受委托限额 120 次/分、400 次/小时、5 并发 且 429 缺 Retry-After 约束（限流实测），Worker 限制 3 并发 + 双层令牌桶（90 次/分、360 次/小时）+ 指数退避（30s→300s），每页落盘 cursor_state 实现断点续传。
  • 通道 B（本地 .onepkg 高速解析）：Windows OneNote 导出 *.onepkg 后拖入 PC 端，Rust 调用 onenote.rs 本地直读二进制分区文件，绕过云端限流，数千页笔记 5–10 分钟即可迁完。
  • HTML 清洗与 AST 转换：将 data-absolute-enabled 绝对定位容器（输出 HTML 规范）按 (top, left) 线性化；依据页面 level（0/1/2）还原父子嵌套；图片转存 S3；内嵌 .docx/.xlsx 自动拆分为子节点并在正文插引用卡片；最终编译为 Y.Doc 二进制入库。
